// --- MOTOR DE AVATAR REFORÇADO --- const avatarImg = document.getElementById('avatar-img'); window.sortearAvatar = () => { // Adicionamos um timestamp (Date.now()) para forçar o navegador a buscar uma nova imagem const seed = Math.random().toString(36).substring(7); const url = `https://api.dicebear.com/7.x/bottts/svg?seed=${seed}&t=${Date.now()}`; avatarImg.src = url; localStorage.setItem('avatarUrl', url); console.log("Avatar atualizado para:", url); }; // Ao carregar a página, se não houver um avatar, sorteia um automaticamente if (!localStorage.getItem('avatarUrl')) { sortearAvatar(); } else { avatarImg.src = localStorage.getItem('avatarUrl'); }
